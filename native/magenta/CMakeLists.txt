cmake_minimum_required(VERSION 3.27)
project(dance2music_magenta LANGUAGES C CXX OBJCXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  message(FATAL_ERROR "The pinned MRT2 native engine requires Apple Silicon macOS.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(FetchContent)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "Pinned commits need no remote update during rebuild")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/manifest.lock.json" lock)
string(JSON commit GET "${lock}" upstream commit)
string(JSON hash GET "${lock}" upstream archive_sha256)
FetchContent_Declare(mrt2 URL "https://codeload.github.com/magenta/magenta-realtime/tar.gz/${commit}" URL_HASH "SHA256=${hash}")
FetchContent_Populate(mrt2)

# Build the upstream core only: no app bundles, JS toolchain, or audio device.
string(JSON dependency_count LENGTH "${lock}" dependencies)
math(EXPR dependency_last "${dependency_count}-1")
foreach(i RANGE 0 ${dependency_last})
  string(JSON name GET "${lock}" dependencies ${i} name)
  string(JSON url GET "${lock}" dependencies ${i} url)
  string(JSON hash GET "${lock}" dependencies ${i} sha256)
  FetchContent_Declare(${name} URL "${url}" URL_HASH "SHA256=${hash}")
endforeach()
string(JSON git_count LENGTH "${lock}" transitive_git)
math(EXPR git_last "${git_count}-1")
foreach(i RANGE 0 ${git_last})
  string(JSON name GET "${lock}" transitive_git ${i} name)
  string(JSON repository GET "${lock}" transitive_git ${i} repository)
  string(JSON commit GET "${lock}" transitive_git ${i} commit)
  FetchContent_Declare(${name} GIT_REPOSITORY "${repository}" GIT_TAG "${commit}" GIT_SUBMODULES "")
endforeach()
foreach(flag TESTS EXAMPLES BENCHMARKS PYTHON_BINDINGS PYTHON_STUBS GGUF CUDA)
  set(MLX_BUILD_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ABSL_INTERNAL_AT_LEAST_CXX20 ON CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON)
set(TFLITE_ENABLE_XNNPACK OFF CACHE BOOL "" FORCE)
set(TFLITE_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(TFLITE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
# MLX first resolves its colocated shader library. Keep the fallback relocatable
# too; never bake a developer's build tree into a distributed executable.
set(MLX_METAL_PATH "." CACHE STRING "Relative Metal fallback" FORCE)
FetchContent_MakeAvailable(mlx sentencepiece)
FetchContent_Populate(tensorflow)
set(FETCHCONTENT_SOURCE_DIR_TENSORFLOW "${tensorflow_SOURCE_DIR}" CACHE PATH "" FORCE)
set(TENSORFLOW_SOURCE_DIR "${tensorflow_SOURCE_DIR}" CACHE PATH "" FORCE)
add_subdirectory("${tensorflow_SOURCE_DIR}/tensorflow/lite" "${CMAKE_BINARY_DIR}/tflite" EXCLUDE_FROM_ALL)
# An additive API bridge exports completed, genuine MusicCoCa embeddings. Keep
# encoding in a separate engine so the live runner can still apply weights
# while a slow audio reference is being encoded. Do not alter model inference.
file(COPY "${mrt2_SOURCE_DIR}/core" DESTINATION "${CMAKE_BINARY_DIR}/mrt2-patched")
foreach(file include/magentart/mlx_engine.h src/mlx_engine.cpp include/magentart/realtime_runner.h src/realtime_runner.cpp)
  configure_file("${mrt2_SOURCE_DIR}/core/${file}" "${CMAKE_BINARY_DIR}/mrt2-patched/core/${file}" COPYONLY)
endforeach()
execute_process(COMMAND /usr/bin/patch --batch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/embedding.patch"
  WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/mrt2-patched" RESULT_VARIABLE patch_result)
if(NOT patch_result EQUAL 0)
  message(FATAL_ERROR "Cannot apply the pinned MRT2 embedding bridge")
endif()
add_subdirectory("${CMAKE_BINARY_DIR}/mrt2-patched/core" "${CMAKE_BINARY_DIR}/mrt2-core" EXCLUDE_FROM_ALL)
add_executable(mrt2_probe probe.cpp)
target_link_libraries(mrt2_probe PRIVATE magentart::core)
add_executable(mrt2_helper helper.cpp)
target_link_libraries(mrt2_helper PRIVATE magentart::core nlohmann_json::nlohmann_json)
# Both developer and packaged launches use the same colocated shader lookup.
# Refresh it even when the executables are up to date, so preparation repairs
# a missing or corrupt runtime copy without needing to relink the helper.
add_custom_target(mrt2_metallib
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${mlx_BINARY_DIR}/mlx/backend/metal/kernels/mlx.metallib"
    "$<TARGET_FILE_DIR:mrt2_helper>/mlx.metallib")
add_dependencies(mrt2_metallib mlx-metallib)
foreach(target mrt2_helper mrt2_probe)
  add_dependencies(${target} mrt2_metallib)
endforeach()
